Summary

Summary: Analysis of the flow cytometry data gathered 2026-02-17, on episomal Ssd1 RNA-binding point mutants, in a Sun4-mNeonGreen reporter background.

This script only includes code to generate figures for the paper. Additional code is in 2026-02-17_Ssd1_pm_redo or similar scripts.

Load packages

Define functions to load flow cytometry data, using FlowCore package

repair_flow_colnames <- function(string) {
  string %>%
    stringr::str_replace_all(" ", "") %>%
    stringr::str_replace_all("PE-", "PE") %>%
    stringr::str_replace_all("-", "_")
}

read_fcs_tibble <- function(file,...) {
  read.FCS(file,...) %>%
    .@exprs %>%
    as_tibble %>%
    set_names(repair_flow_colnames(names(.)))
}

Calculate fold changes between the strain means

mRuby2, comparing all strains to the empty plasmid strain

# Set Empty plasmid strain as the reference

flow_data_gated_buddedsinglets_emptyref <- flow_data_gated_buddedsinglets %>%
  dplyr::filter(`610_20(561)_A` > 0) %>%
  dplyr::ungroup() %>%
  dplyr::mutate(ssd1locus = relevel(as.factor(ssd1locus), ref = "Empty"))

# log10 linear model

fit_mRuby2 <- lm(
  log10(`610_20(561)_A`) ~ ssd1locus,
  data = flow_data_gated_buddedsinglets_emptyref)

foldchange_mRuby2 <- 10^{ coef(fit_mRuby2)[-1] }
